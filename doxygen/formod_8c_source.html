63int main(
64 int argc,
65 char *argv[]) {
66
67 static ctl_t ctl;
68
69 /* Print usage information... */
70 USAGE;
71
72 /* Check arguments... */
73 if (argc < 5)
74 ERRMSG("Missing or invalid command-line arguments.\n\n"
75 "Usage: formod <ctl> <obs> <atm> <rad> [KEY VALUE ...]\n\n"
76 "Use -h for full help.");
77
78 /* Read control parameters... */
79 SELECT_TIMER("READ_CTL", "INPUT");
80 read_ctl(argc, argv, &ctl);
81
82#ifdef UNIFIED
83
84 static atm_t atm;
85 static obs_t obs;
86
87 /* Read observation geometry... */
88 SELECT_TIMER("READ_OBS", "INPUT");
89 read_obs(NULL, argv[2], &ctl, &obs, 0);
90
91 /* Read atmospheric data... */
92 SELECT_TIMER("READ_ATM", "INPUT");
93 read_atm(NULL, argv[3], &ctl, &atm, 0);
94
95 /* Call forward model... */
96 SELECT_TIMER("FORMOD_UNIFIED", "FORWARD");
97 jur_unified_init(argc, argv);
98 jur_unified_formod_multiple_packages(&atm, &obs, 1, NULL);
99
100 /* Save radiance data... */
101 SELECT_TIMER("WRITE_OBS", "OUTPUT");
102 write_obs(NULL, argv[4], &ctl, &obs, 0);
103
104#else
105
106 char dirlist[LEN], obsref[LEN], task[LEN];
107
108 /* Initialize look-up tables... */
109 SELECT_TIMER("READ_TBL", "INPUT");
110 tbl_t *tbl = read_tbl(&ctl);
111
112 /* Get task... */
113 SELECT_TIMER("READ_TASK", "INPUT");
114 scan_ctl(argc, argv, "TASK", -1, "-", task);
115
116 /* Get dirlist... */
117 SELECT_TIMER("READ_DIRLIST", "INPUT");
118 scan_ctl(argc, argv, "DIRLIST", -1, "-", dirlist);
119
120 /* Get reference data... */
121 SELECT_TIMER("READ_OBSREF", "INPUT");
122 scan_ctl(argc, argv, "OBSREF", -1, "-", obsref);
123
124 /* Single forward calculation... */
125 if (dirlist[0] == '-')
126 call_formod(&ctl, tbl, NULL, argv[2], argv[3], argv[4], task, obsref);
127
128 /* Work on directory list... */
129 else {
130
131 /* Open directory list... */
132 FILE *in;
133 if (!(in = fopen(dirlist, "r")))
134 ERRMSG("Cannot open directory list!");
135
136 /* Loop over directories... */
137 char wrkdir[LEN];
138 while (fscanf(in, "%4999s", wrkdir) != EOF) {
139
140 /* Write info... */
141 LOG(1, "\nWorking directory: %s", wrkdir);
142
143 /* Call forward model... */
144 call_formod(&ctl, tbl, wrkdir, argv[2], argv[3], argv[4], task, obsref);
145 }
146
147 /* Close dirlist... */
148 fclose(in);
149 }
150
151#endif
152
153 /* Free... */
154 SELECT_TIMER("FINALIZE", "OVERHEAD");
155 tbl_free(&ctl, tbl);
156 PRINT_TIMERS;
157
158 return EXIT_SUCCESS;
159}

211void call_formod(
212 ctl_t *ctl,
213 const tbl_t *tbl,
214 const char *wrkdir,
215 const char *obsfile,
216 const char *atmfile,
217 const char *radfile,
218 const char *task,
219 const char *obsref) {
220
221 static atm_t atm, atm2;
222 static obs_t obs, obs2;
223
224 /* Read atmospheric data... */
225 SELECT_TIMER("READ_ATM", "INPUT");
226 read_atm(wrkdir, atmfile, ctl, &atm, 0);
227
228 /* Read observation geometry... */
229 SELECT_TIMER("READ_OBS", "INPUT");
230 read_obs(wrkdir, obsfile, ctl, &obs, 0);
231
232 /* Compute multiple profiles... */
233 if (task[0] == 'p' || task[0] == 'P') {
234
235 SELECT_TIMER("FORMOD_PROFILE_LOOP", "FORWARD");
236
237 /* Loop over ray paths... */
238 for (int ir = 0; ir < obs.nr; ir++) {
239
240 /* Get atmospheric data... */
241 atm2.np = 0;
242 for (int ip = 0; ip < atm.np; ip++)
243 if (atm.time[ip] == obs.time[ir]) {
244 atm2.time[atm2.np] = atm.time[ip];
245 atm2.z[atm2.np] = atm.z[ip];
246 atm2.lon[atm2.np] = atm.lon[ip];
247 atm2.lat[atm2.np] = atm.lat[ip];
248 atm2.p[atm2.np] = atm.p[ip];
249 atm2.t[atm2.np] = atm.t[ip];
250 for (int ig = 0; ig < ctl->ng; ig++)
251 atm2.q[ig][atm2.np] = atm.q[ig][ip];
252 for (int iw = 0; iw < ctl->nw; iw++)
253 atm2.k[iw][atm2.np] = atm.k[iw][ip];
254 atm2.np++;
255 }
256
257 /* Get observation data... */
258 obs2.nr = 1;
259 obs2.time[0] = obs.time[ir];
260 obs2.vpz[0] = obs.vpz[ir];
261 obs2.vplon[0] = obs.vplon[ir];
262 obs2.vplat[0] = obs.vplat[ir];
263 obs2.obsz[0] = obs.obsz[ir];
264 obs2.obslon[0] = obs.obslon[ir];
265 obs2.obslat[0] = obs.obslat[ir];
266
267 /* Check number of data points... */
268 if (atm2.np > 0) {
269
270 /* Call forward model... */
271 formod(ctl, tbl, &atm2, &obs2);
272
273 /* Save radiance data... */
274 for (int id = 0; id < ctl->nd; id++) {
275 obs.rad[id][ir] = obs2.rad[id][0];
276 obs.tau[id][ir] = obs2.tau[id][0];
277 }
278 }
279 }
280
281 /* Write radiance data... */
282 SELECT_TIMER("WRITE_OBS", "OUTPUT");
283 write_obs(wrkdir, radfile, ctl, &obs, 0);
284 }
285
286 /* Compute single profile... */
287 else {
288
289 /* Call forward model... */
290 SELECT_TIMER("FORMOD", "FORWARD");
291 formod(ctl, tbl, &atm, &obs);
292
293 /* Save radiance data... */
294 SELECT_TIMER("WRITE_OBS", "OUTPUT");
295 write_obs(wrkdir, radfile, ctl, &obs, 0);
296
297 /* Evaluate results... */
298 if (obsref[0] != '-') {
299
300 SELECT_TIMER("EVAL", "OUTPUT");
301
302 /* Read reference data... */
303 read_obs(wrkdir, obsref, ctl, &obs2, 0);
304
305 /* Calculate relative errors... */
306 double mre[ND], sdre[ND], minre[ND], maxre[ND];
307 compute_rel_errors(ctl, &obs, &obs2, mre, sdre, minre, maxre);
308
309 /* Write results... */
310 for (int id = 0; id < ctl->nd; id++)
311 printf
312 ("EVAL: nu= %.4f cm^-1 | MRE= %g %% | SDRE= %g %% | MinRE= %g %% | MaxRE= %g %%\n",
313 ctl->nu[id], mre[id], sdre[id], minre[id], maxre[id]);
314 }
315
316 /* Compute contributions of emitters... */
317 if (task[0] == 'c' || task[0] == 'C') {
318
319 SELECT_TIMER("CONTRIBUTIONS", "FORWARD");
320
321 char filename[LEN];
322
323 /* Switch off continua... */
324 ctl->ctm_co2 = 0;
325 ctl->ctm_h2o = 0;
326 ctl->ctm_n2 = 0;
327 ctl->ctm_o2 = 0;
328
329 /* Loop over emitters... */
330 for (int ig = 0; ig < ctl->ng; ig++) {
331
332 /* Copy atmospheric data... */
333 copy_atm(ctl, &atm2, &atm, 0);
334
335 /* Set extinction to zero... */
336 for (int iw = 0; iw < ctl->nw; iw++)
337 for (int ip = 0; ip < atm2.np; ip++)
338 atm2.k[iw][ip] = 0;
339
340 /* Select emitter... */
341 for (int ig2 = 0; ig2 < ctl->ng; ig2++)
342 if (ig2 != ig)
343 for (int ip = 0; ip < atm2.np; ip++)
344 atm2.q[ig2][ip] = 0;
345
346 /* Call forward model... */
347 formod(ctl, tbl, &atm2, &obs);
348
349 /* Save radiance data... */
350 sprintf(filename, "%s.%s", radfile, ctl->emitter[ig]);
351 write_obs(wrkdir, filename, ctl, &obs, 0);
352 }
353
354 /* Copy atmospheric data... */
355 copy_atm(ctl, &atm2, &atm, 0);
356
357 /* Set volume mixing ratios to zero, keep extinction... */
358 for (int ig = 0; ig < ctl->ng; ig++)
359 for (int ip = 0; ip < atm2.np; ip++)
360 atm2.q[ig][ip] = 0;
361
362 /* Call forward model... */
363 formod(ctl, tbl, &atm2, &obs);
364
365 /* Save radiance data... */
366 sprintf(filename, "%s.EXTINCT", radfile);
367 write_obs(wrkdir, filename, ctl, &obs, 0);
368 }
369
370 /* Measure CPU-time... */
371 if (task[0] == 't' || task[0] == 'T') {
372
373 /* Init... */
374 double t_min = 0, t_max = 0, t_mean = 0, t_sd = 0;
375 int n = 0;
376
377 /* Initialize random number generator... */
378 gsl_rng_env_setup();
379 gsl_rng *rng = gsl_rng_alloc(gsl_rng_default);
380
381 /* Loop over profiles... */
382 do {
383
384 /* Create random atmosphere... */
385 copy_atm(ctl, &atm2, &atm, 0);
386 double dtemp = 40. * (gsl_rng_uniform(rng) - 0.5);
387 double dpress = 1. - 0.1 * gsl_rng_uniform(rng);
388 double dq[NG];
389 for (int ig = 0; ig < ctl->ng; ig++)
390 dq[ig] = 0.8 + 0.4 * gsl_rng_uniform(rng);
391 for (int ip = 0; ip < atm2.np; ip++) {
392 atm2.t[ip] += dtemp;
393 atm2.p[ip] *= dpress;
394 for (int ig = 0; ig < ctl->ng; ig++)
395 atm2.q[ig][ip] *= dq[ig];
396 }
397
398 /* Measure runtime... */
399 SELECT_TIMER("BENCHMARK_SAMPLE", "ANALYSIS");
400 double t0 = omp_get_wtime();
401 formod(ctl, tbl, &atm2, &obs);
402 double dt = omp_get_wtime() - t0;
403
404 /* Get runtime statistics... */
405 t_mean += dt;
406 t_sd += POW2(dt);
407 if (n == 0 || dt < t_min)
408 t_min = dt;
409 if (n == 0 || dt > t_max)
410 t_max = dt;
411 n++;
412
413 } while (t_mean < 10.0);
414
415 /* Keep the legacy benchmark line and the new timer-style overlap metric. */
416 t_mean /= (double) n;
417 t_sd = sqrt(t_sd / (double) n - POW2(t_mean));
418 printf("RUNTIME: mean= %g s | stddev= %g s | min= %g s | max= %g s\n",
419 t_mean, t_sd, t_min, t_max);
420
421 /* Free... */
422 gsl_rng_free(rng);
423 }
424
425 /* Analyze impact of step size... */
426 if (task[0] == 's' || task[0] == 'S') {
427
428 SELECT_TIMER("STEP_ANALYSIS", "ANALYSIS");
429
430 /* Reference run... */
431 ctl->rayds = 0.1;
432 ctl->raydz = 0.01;
433 formod(ctl, tbl, &atm, &obs);
434 copy_obs(ctl, &obs2, &obs, 0);
435
436 /* Loop over step size... */
437 for (double dz = 0.01; dz <= 2; dz *= 1.1)
438 for (double ds = 0.1; ds <= 50; ds *= 1.1) {
439
440 /* Set step size... */
441 ctl->rayds = ds;
442 ctl->raydz = dz;
443
444 /* Measure runtime... */
445 double t0 = omp_get_wtime();
446 formod(ctl, tbl, &atm, &obs);
447 double dt = omp_get_wtime() - t0;
448
449 /* Calculate relative errors... */
450 double mre[ND], sdre[ND], minre[ND], maxre[ND];
451 compute_rel_errors(ctl, &obs, &obs2, mre, sdre, minre, maxre);
452
453 /* Write results... */
454 for (int id = 0; id < ctl->nd; id++)
455 printf
456 ("STEPSIZE: ds= %.4f km | dz= %g km | t= %g s | nu= %.4f cm^-1"
457 " | MRE= %g %% | SDRE= %g %% | MinRE= %g %% | MaxRE= %g %%\n",
458 ds, dz, dt, ctl->nu[id], mre[id], sdre[id], minre[id],
459 maxre[id]);
460 }
461 }
462 }
463}

465/*****************************************************************************/
466
467void compute_rel_errors(
468 const ctl_t *ctl,
469 const obs_t *obs_test,
470 const obs_t *obs_ref,
471 double *mre,
472 double *sdre,
473 double *minre,
474 double *maxre) {
475
476 /* Loop over channels... */
477 for (int id = 0; id < ctl->nd; id++) {
478
479 double sum = 0, sum2 = 0;
480 minre[id] = +1e9;
481 maxre[id] = -1e9;
482 int n = 0;
483
484 /* Loop over ray paths... */
485 for (int ir = 0; ir < obs_test->nr; ir++) {
486
487 /* Check for zero... */
488 if (obs_ref->rad[id][ir] == 0)
489 continue;
490
491 /* Calculate relative error... */
492 double err = 100.0 * (obs_test->rad[id][ir] - obs_ref->rad[id][ir])
493 / obs_ref->rad[id][ir];
494
495 /* Get statistics... */
496 sum += err;
497 sum2 += err * err;
498 if (err > maxre[id])
499 maxre[id] = err;
500 if (err < minre[id])
501 minre[id] = err;
502 n++;
503 }
504
505 /* Get mean and standard deviaton... */
506 mre[id] = sum / n;
507 sdre[id] = sqrt(sum2 / n - mre[id] * mre[id]);
508 }
509}
